    function getAccountAlertConfig(account, currentBalanceOverride = null) {
      const stored = getStoredAccountAlerts()[account.id] || {};
      const minBalance = parseFloat(account.min_balance_alert ?? stored.min_balance_alert ?? account.min_balance ?? NaN);
      const maxBalance = parseFloat(account.max_balance_alert ?? stored.max_balance_alert ?? account.max_balance ?? NaN);
      const signedBalance = Number.isFinite(currentBalanceOverride) ? currentBalanceOverride : calculateAccountBalance(account);
      const isCreditAccount = String(account.type || '').toLowerCase() === 'credit';
      const debtBalance = isCreditAccount ? Math.abs(signedBalance) : signedBalance;

      return {
        minBalance: Number.isFinite(minBalance) ? minBalance : null,
        maxBalance: Number.isFinite(maxBalance) ? maxBalance : null,
        minEnabled: Number.isFinite(minBalance),
        maxEnabled: Number.isFinite(maxBalance),
        signedBalance,
        debtBalance,
        isCreditAccount
      };
    }

    function calculateAccountBalance(account, txListOverride = null) {
      const initialBal = parseFloat(account?.initial_balance || 0);
      let txList = txListOverride;

      if (!txList) {
        const globalTxs = (typeof transactions !== 'undefined' && transactions.length > 0)
          ? transactions.filter(tx => (tx.account_id === account.id || tx.accountId === account.id))
          : [];

        txList = globalTxs.length > 0 ? globalTxs : (Array.isArray(account.transactions) ? account.transactions : []);
      }

      const totalTxSum = txList.reduce((sum, tx) => sum + parseFloat(tx.amount || 0), 0);
      return initialBal + totalTxSum;
    }

    function isTransferPartnerTransaction(tx) {
      return !!(tx && tx.transfer_id);
    }

    function getSelectedMonthTransactions() {
      const [year, month] = selectedSummaryMonth.split('-').map(Number);
      if (!year || !month) return [];

      const monthStart = new Date(year, month - 1, 1);
      const monthEnd = new Date(year, month, 0, 23, 59, 59, 999);

      return (Array.isArray(transactions) ? transactions : []).filter(tx => {
        const txDate = tx && tx.date ? new Date(tx.date + 'T00:00:00') : null;
        if (!txDate || Number.isNaN(txDate.getTime())) return false;
        if (isTransferPartnerTransaction(tx)) return false;
        return txDate >= monthStart && txDate <= monthEnd;
      });
    }

    function compareTransactions(a, b) {
      const statusA = a?.status === 'pending' ? 0 : 1;
      const statusB = b?.status === 'pending' ? 0 : 1;
      if (statusA !== statusB) return statusA - statusB;

      const dateA = a?.date ? new Date(`${a.date}T00:00:00`).getTime() : 0;
      const dateB = b?.date ? new Date(`${b.date}T00:00:00`).getTime() : 0;
      if (dateA !== dateB) return dateB - dateA;

      const createdA = a?.created_at ? new Date(a.created_at).getTime() : 0;
      const createdB = b?.created_at ? new Date(b.created_at).getTime() : 0;
      if (createdA !== createdB) return createdB - createdA;

      if (a?.transfer_id && b?.transfer_id && a.transfer_id === b.transfer_id) return 0;
      return (a?.transfer_id && !b?.transfer_id) ? -1 : (!a?.transfer_id && b?.transfer_id) ? 1 : 0;
    }
