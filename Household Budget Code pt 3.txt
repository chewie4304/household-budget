    function populateTargetAccountDropdown() {
      const targetSelect = document.getElementById('txTargetAccount');
      const sourceAccountId = document.getElementById('txAccount').value;

      if (!targetSelect || !Array.isArray(accounts)) return;

      targetSelect.innerHTML = accounts
        .filter(acc => acc.id !== sourceAccountId)
        .map(acc => `<option value="${acc.id}">${acc.name}</option>`)
        .join('');
    }

    function populateAccountDropdown(selectedId = null) {
      const select = document.getElementById('txAccount');
      if (!accounts || accounts.length === 0) {
        select.innerHTML = `<option value="">No accounts available</option>`;
        return;
      }

      select.innerHTML = accounts.map(acc => {
        const isSelected = (selectedId && acc.id === selectedId) || (!selectedId && activeAccountId && acc.id === activeAccountId);
        return `<option value="${acc.id}" ${isSelected ? 'selected' : ''}>${acc.name} (${acc.type})</option>`;
      }).join('');
    }

    // --- Image Cropper Integration Logic ---
    function handleImageInput(input) {
      const file = input.files && input.files[0];
      if (!file) return;

      const isImage = file.type && file.type.startsWith('image/');
      const isPdf = file.type === 'application/pdf' || file.name.toLowerCase().endsWith('.pdf');

      if (!isImage && !isPdf) {
        showAppMessage('Please choose an image or PDF file for the receipt.');
        input.value = '';
        return;
      }

      if (isPdf) {
        activeReceiptFiles.push(file);
        input.value = '';
        renderModalReceiptList();
        return;
      }

      const reader = new FileReader();
      reader.onload = (e) => {
        const imageToCrop = document.getElementById('imageToCrop');
        imageToCrop.src = e.target.result;
        document.getElementById('cropModal').classList.remove('hidden');

        if (cropper) {
          cropper.destroy();
        }

        cropper = new Cropper(imageToCrop, {
          aspectRatio: NaN,
          viewMode: 1,
        });
      };
      reader.readAsDataURL(file);
      input.value = '';
    }

    function cancelCrop() {
      document.getElementById('cropModal').classList.add('hidden');
      if (cropper) {
        cropper.destroy();
        cropper = null;
      }
    }

    function confirmCrop() {
      if (!cropper) return;

      const canvas = cropper.getCroppedCanvas({
        maxWidth: 1200,
        maxHeight: 1200,
        imageSmoothingQuality: 'high',
      });

      canvas.toBlob((blob) => {
        const croppedFile = new File([blob], `receipt_${Date.now()}.jpg`, {type: 'image/jpeg'});
        activeReceiptFiles.push(croppedFile);

        cancelCrop();
        renderModalReceiptList();
      }, 'image/jpeg', 0.85);
    }

    function renderModalReceiptList() {
      const preview = document.getElementById('receiptPreview');
      const container = document.getElementById('receiptLinksList');
      const statusText = document.getElementById('fileSelectionStatus');

      const totalCount = activeReceiptUrls.length + activeReceiptFiles.length;

      if (totalCount === 0) {
        preview.classList.add('hidden');
        container.innerHTML = '';
        statusText.classList.add('hidden');
        return;
      }

      preview.classList.remove('hidden');
      statusText.textContent = `✓ ${totalCount} attachment${totalCount > 1 ? 's' : ''} ready`;
      statusText.classList.remove('hidden');

      let html = '';

      activeReceiptUrls.forEach((url, idx) => {
        html += `
          <div class="flex items-center justify-between bg-gray-50 border border-gray-200 p-2 rounded-lg text-xs">
            <a href="${url}" target="_blank" class="text-blue-600 font-semibold underline truncate max-w-[200px]">
              Receipt URL ${activeReceiptUrls.length > 1 ? idx + 1 : ''}
            </a>
            <button type="button" onclick="removeExistingReceipt(${idx})" class="bg-red-50 hover:bg-red-100 text-red-600 border border-red-200 text-[11px] font-semibold px-2 py-0.5 rounded transition-colors">
              Remove
            </button>
          </div>
        `;
      });

      activeReceiptFiles.forEach((file, idx) => {
        const fileLabel = file.type === 'application/pdf' || file.name.toLowerCase().endsWith('.pdf')
          ? `📄 PDF: ${file.name}`
          : `📷 Cropped: ${file.name}`;
        html += `
          <div class="flex items-center justify-between bg-blue-50 border border-blue-200 p-2 rounded-lg text-xs">
            <span class="text-blue-700 font-semibold truncate max-w-[200px]">
              ${fileLabel}
            </span>
            <button type="button" onclick="removeNewReceipt(${idx})" class="bg-red-50 hover:bg-red-100 text-red-600 border border-red-200 text-[11px] font-semibold px-2 py-0.5 rounded transition-colors">
              Remove
            </button>
          </div>
        `;
      });

      container.innerHTML = html;
    }

    function removeExistingReceipt(index) {
      activeReceiptUrls.splice(index, 1);
      renderModalReceiptList();
    }

    function removeNewReceipt(index) {
      activeReceiptFiles.splice(index, 1);
      renderModalReceiptList();
    }

    async function uploadReceiptFiles() {
      const uploadedUrls = [];

      for (const file of activeReceiptFiles) {
        const safeName = file.name.replace(/[^a-zA-Z0-9._-]/g, '_');
        const path = `${crypto.randomUUID()}-${safeName}`;
        const {error: uploadError} = await sb.storage
          .from(RECEIPTS_BUCKET)
          .upload(path, file, {contentType: file.type, upsert: false});

        if (uploadError) {
          throw new Error(`Could not upload ${file.name}: ${uploadError.message}`);
        }

        const {data} = sb.storage.from(RECEIPTS_BUCKET).getPublicUrl(path);
        if (!data?.publicUrl) {
          throw new Error(`Could not create a URL for ${file.name}.`);
        }
        uploadedUrls.push(data.publicUrl);
      }

      return uploadedUrls;
    }

    function toggleCheckNumberField() {
      const paymentSelect = document.getElementById('txPaymentMethod');
      const checkWrapper = document.getElementById('checkNumberWrapper');
      const checkInput = document.getElementById('txCheckNumber');

      if (!paymentSelect || !checkWrapper || !checkInput) return;

      const isCheck = paymentSelect.value === 'check' && supportsCheckNumber;
      checkWrapper.classList.toggle('hidden', !isCheck);
      if (!isCheck) {
        checkInput.value = '';
      }
    }

    function autoSelectPaymentMethod() {
      const accountSelect = document.getElementById('txAccount');
      const paymentSelect = document.getElementById('txPaymentMethod');
      if (!accountSelect || !paymentSelect) return;

      const selectedAccountId = accountSelect.value;
      const selectedAccount = accounts.find(a => String(a.id) === String(selectedAccountId));
      if (!selectedAccount) return;

      const nameAndType = `${selectedAccount.name || ''} ${selectedAccount.type || ''}`.toLowerCase();

      if (nameAndType.includes('credit') || nameAndType.includes('visa') || nameAndType.includes('mastercard') || nameAndType.includes('card')) {
        paymentSelect.value = 'credit';
      } else if (nameAndType.includes('checking') || nameAndType.includes('debit')) {
        paymentSelect.value = 'debit';
      } else if (nameAndType.includes('cash')) {
        paymentSelect.value = 'cash';
      } else if (nameAndType.includes('savings')) {
        paymentSelect.value = 'transfer';
      }

      toggleCheckNumberField();
    }

    function getLocalDateString(date = new Date()) {
      const year = date.getFullYear();
      const month = String(date.getMonth() + 1).padStart(2, '0');
      const day = String(date.getDate()).padStart(2, '0');
      return `${year}-${month}-${day}`;
    }

    function updateAutocompleteSuggestions() {
      const payeeList = document.getElementById('payee-suggestions');
      const descList = document.getElementById('desc-suggestions');

      if (!payeeList || !descList || !Array.isArray(transactions)) return;

      // Extract unique payees (ignoring empty ones)
      const uniquePayees = [...new Set(
        transactions
          .map(tx => (tx.payee || '').trim())
          .filter(p => p.length > 0)
      )].sort((a, b) => a.localeCompare(b, undefined, {sensitivity: 'base'}));

      // Extract unique descriptions (ignoring empty ones)
      const uniqueDescs = [...new Set(
        transactions
          .map(tx => (tx.description || '').trim())
          .filter(d => d.length > 0)
      )].sort((a, b) => a.localeCompare(b, undefined, {sensitivity: 'base'}));

      // Build options HTML
      payeeList.innerHTML = uniquePayees.map(p => `<option value="${p}">`).join('');
      descList.innerHTML = uniqueDescs.map(d => `<option value="${d}">`).join('');
    }

    function openTxModal(txId = null) {
      if (accounts.length === 0) {
        showAppMessage("Please create an account first before adding transactions.");
        return;
      }

      const form = document.getElementById('tx-form');
      form.reset();
      const deleteBtn = document.getElementById('txDeleteBtn');

      activeReceiptFiles = [];
      activeReceiptUrls = [];
      document.getElementById('txSplitEnabled').checked = false;
      document.getElementById('txSplitEnabled').disabled = false;
      renderSplitEditor();

      if (txId) {
        const tx = transactions.find(t => t.id === txId);
        if (!tx) return;

        const amt = parseFloat(tx.amount || 0);
        document.getElementById('tx-modal-title').textContent = "Edit Transaction";
        document.getElementById('txSubmitBtn').textContent = "Update Transaction";
        document.getElementById('editingTxId').value = tx.id;
        populateAccountDropdown(tx.account_id);
        document.getElementById('txDesc').value = tx.description || '';
        document.getElementById('txPayee').value = tx.payee || '';
        document.getElementById('txType').value = amt >= 0 ? 'income' : 'expense';
        document.getElementById('txAmount').value = Math.abs(amt);
        document.getElementById('txPaymentMethod').value = tx.payment_method || 'debit';
        document.getElementById('txCheckNumber').value = tx.check_number || '';
        document.getElementById('txDate').value = tx.date || getLocalDateString();
        document.getElementById('txStatus').value = tx.status || 'pending';
        document.getElementById('txTags').value = Array.isArray(tx.tags) ? tx.tags.join(', ') : '';

        const existingSplits = getSplitDetails(tx);
        if (existingSplits.length > 0) {
          document.getElementById('txSplitEnabled').checked = true;
          renderSplitEditor(existingSplits);
        }

        if (tx.receipt_url) {
          activeReceiptUrls = tx.receipt_url.split(',').map(u => u.trim()).filter(Boolean);
        }

        deleteBtn.classList.remove('hidden');
      } else {
        document.getElementById('tx-modal-title').textContent = "Add Transaction";
        document.getElementById('txSubmitBtn').textContent = "Save Transaction";
        document.getElementById('editingTxId').value = "";
        populateAccountDropdown(activeAccountId);

        // Automatically match payment method to the default/active account
        autoSelectPaymentMethod();

        document.getElementById('txPayee').value = "";
        document.getElementById('txDate').value = getLocalDateString();
        document.getElementById('txStatus').value = 'pending';
        document.getElementById('txTags').value = '';

        deleteBtn.classList.add('hidden');
      }

      renderModalReceiptList();
      updatePayeeLabel();
      updateAutocompleteSuggestions();
      toggleCheckNumberField();
      handleTypeChange();
      toggleTxModal(true);
    }

    async function fetchData() {
      // Try ordering by created_at or name if 'id' isn't sequential
      const {data: accData, error: accError} = await sb.from('accounts').select('*').order('created_at', {ascending: true});
      const {data: txData, error: txError} = await sb.from('transactions').select('*').order('date', {ascending: false});

      if (accError) console.error("Error fetching accounts:", accError);
      if (txError) console.error("Error fetching transactions:", txError);

      if (accData) accounts = accData;
      if (txData) transactions = txData;

      // Re-render active view
      if (window.currentAccountId) {
        showTransactionsView(window.currentAccountId);
      } else {
        renderAccounts();
      }
    }
