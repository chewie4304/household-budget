    async function handleTxSubmit(e) {
      e.preventDefault();
      const submitBtn = document.getElementById('txSubmitBtn');
      submitBtn.disabled = true;
      submitBtn.textContent = "Saving...";

      const editId = document.getElementById('editingTxId').value;
      const targetAccountId = document.getElementById('txAccount').value;
      const destinationAccountId = document.getElementById('txTargetAccount')?.value;
      const description = document.getElementById('txDesc').value;
      const payee = document.getElementById('txPayee').value;
      const savedPayee = payee || 'Transaction';
      const payment_method = document.getElementById('txPaymentMethod').value;
      const shouldStoreCheckNumber = payment_method === 'check' && supportsCheckNumber;
      const check_number = shouldStoreCheckNumber ? document.getElementById('txCheckNumber').value.trim() : '';
      const txType = document.getElementById('txType').value;
      let amount = parseFloat(document.getElementById('txAmount').value) || 0;
      const date = document.getElementById('txDate').value;
      const status = document.getElementById('txStatus').value;
      const tagsRaw = document.getElementById('txTags')?.value || '';
      const tags = parseTagInput(tagsRaw);
      const splitEnabled = document.getElementById('txSplitEnabled')?.checked && txType !== 'transfer';
      const splitDetails = splitEnabled ? getSplitRowsFromEditor() : [];

      if (splitEnabled) {
        const splitTotal = splitDetails.reduce((sum, split) => sum + split.amount, 0);
        if (splitDetails.length < 2 || splitDetails.some(split => split.tags.length === 0 || split.amount <= 0) || Math.abs(splitTotal - Math.abs(amount)) > 0.005) {
          showAppMessage('Each split needs at least one tag, a positive amount, and all split amounts must equal the transaction total.');
          submitBtn.disabled = false;
          submitBtn.textContent = editId ? "Update Transaction" : "Save Transaction";
          return;
        }
      }

      let uploadedReceiptUrls;
      try {
        uploadedReceiptUrls = await uploadReceiptFiles();
      } catch (uploadError) {
        showAppMessage(uploadError.message);
        submitBtn.disabled = false;
        submitBtn.textContent = editId ? "Update Transaction" : "Save Transaction";
        return;
      }
      const finalReceiptUrls = [...activeReceiptUrls, ...uploadedReceiptUrls];

      // Validation for Transfers
      if (txType === 'transfer' && !editId) {
        if (!destinationAccountId || targetAccountId === destinationAccountId) {
          showAppMessage("Please select a different destination account for the transfer.");
          submitBtn.disabled = false;
          submitBtn.textContent = "Save Transaction";
          return;
        }
      }

      const payload = {
        account_id: targetAccountId,
        description: description || '',
        payee: savedPayee,
        payment_method,
        amount,
        date,
        status,
        tags,
        split_details: splitDetails.length > 0 ? splitDetails : null,
        receipt_url: finalReceiptUrls.join(', ')
      };

      if (shouldStoreCheckNumber) {
        payload.check_number = check_number;
      }

      let error;

      if (editId) {
        const adjustedPayload = {
          ...payload,
          amount: txType === 'expense' ? -Math.abs(amount) : Math.abs(amount)
        };
        ({error} = await sb.from('transactions').update(adjustedPayload).eq('id', editId));
      } else if (txType === 'transfer') {
        // Generate a unique link key for both ends of the transfer
        const transferId = crypto.randomUUID();

        const sourceAccount = accounts.find(a => a.id === targetAccountId);
        const targetAccount = accounts.find(a => a.id === destinationAccountId);

        // 1. Debit from source account
        const sourcePayload = {
          ...payload,
          amount: -Math.abs(amount),
          payee: payee || targetAccount?.name || 'Transaction',
          transfer_id: transferId
        };

        if (shouldStoreCheckNumber) {
          sourcePayload.check_number = check_number;
        }

        // 2. Deposit into destination account
        const targetPayload = {
          ...payload,
          account_id: destinationAccountId,
          amount: Math.abs(amount),
          payee: payee || sourceAccount?.name || 'Transaction',
          transfer_id: transferId
        };

        if (shouldStoreCheckNumber) {
          targetPayload.check_number = check_number;
        }

        ({error} = await sb.from('transactions').insert([sourcePayload, targetPayload]));
      } else {
        const adjustedPayload = {
          ...payload,
          amount: txType === 'expense' ? -Math.abs(amount) : Math.abs(amount)
        };
        ({error} = await sb.from('transactions').insert([adjustedPayload]));
      }

      submitBtn.disabled = false;
      submitBtn.textContent = "Save Transaction";

      if (error) {
        showAppMessage("Error saving transaction: " + error.message);
      } else {
        toggleTxModal(false);

        await fetchData();
        renderAccounts();
        renderAllTransactions();

        if (currentViewState === 'home-transactions') {
          showAllTransactions();
        } else if (window.currentAccountId) {
          showTransactionsView(window.currentAccountId);
        } else {
          showHomeAccounts();
        }
      }
    }

    let pendingConfirmAction = null;

      const modal = document.getElementById('confirm-modal');
      pendingConfirmAction = null;
      modal.classList.add('hidden');
    }

      if (typeof pendingConfirmAction === 'function') {
        const action = pendingConfirmAction;
        closeCustomConfirm();

      }

    async function handleDeleteTxFromModal() {
      const id = document.getElementById('editingTxId').value;
      const desc = document.getElementById('txDesc').value || 'this transaction';
      if (!id) return;

      openCustomConfirm({
        title: 'Delete Transaction',
        message: `Delete '${desc}'?`,
        onConfirm: async () => {
          const targetTx = transactions.find(t => t.id == id);
          let query = sb.from('transactions').delete();

          if (targetTx && targetTx.transfer_id) {
            query = query.eq('transfer_id', targetTx.transfer_id);
          } else {
            query = query.eq('id', id);
          }

          const {error} = await query;
          if (error) {
            showAppMessage("Error deleting transaction: " + error.message);
            return;
          }

          toggleTxModal(false);
          await fetchData();
          if (window.currentAccountId) {
            await showTransactionsView(window.currentAccountId);
          } else if (currentViewState === 'home-transactions') {
            showAllTransactions(selectedSummaryTransactionType);
          } else {
            renderAccounts();
          }
        }
      });
    }

      const minEnabled = document.getElementById('accMinAlertEnabled')?.checked;

    function openAccountModal(accountId = null) {
      const form = document.getElementById('account-form');
      form.reset();
      const deleteBtn = document.getElementById('deleteBtn');

      if (accountId) {
        const acc = accounts.find(a => a.id === accountId);
        if (!acc) return;
        const storedAlerts = getStoredAccountAlerts()[acc.id] || {};
        const minBalance = parseFloat(acc.min_balance_alert ?? storedAlerts.min_balance_alert ?? acc.min_balance ?? NaN);
        const maxBalance = parseFloat(acc.max_balance_alert ?? storedAlerts.max_balance_alert ?? acc.max_balance ?? NaN);

        document.getElementById('modal-title').textContent = "Edit Account";
        document.getElementById('submitBtn').textContent = "Update Account";
        document.getElementById('editingAccId').value = acc.id;
        document.getElementById('accName').value = acc.name || '';
        document.getElementById('accType').value = acc.type || 'checking';
        document.getElementById('accBalance').value = acc ? acc.initial_balance : '';
        document.getElementById('accMinAlertEnabled').checked = Number.isFinite(minBalance);
        document.getElementById('accMinBalance').value = Number.isFinite(minBalance) ? minBalance : '';
        document.getElementById('accMaxAlertEnabled').checked = Number.isFinite(maxBalance);
        document.getElementById('accMaxBalance').value = Number.isFinite(maxBalance) ? maxBalance : '';
        document.getElementById('accInstitution').value = acc.institution || '';
        document.getElementById('accNumber').value = acc.account_number || '';
        deleteBtn.classList.remove('hidden');
      } else {
        document.getElementById('modal-title').textContent = "Add New Account";
        document.getElementById('submitBtn').textContent = "Create Account";
        document.getElementById('editingAccId').value = "";
        deleteBtn.classList.add('hidden');
      }

      toggleAccountAlertControls();
      toggleAccountModal(true);
    }

    async function saveAccountData(payload, editId) {
      const supportedColumns = ['name', 'type', 'initial_balance', 'institution', 'account_number'];
      const safePayload = Object.fromEntries(
        Object.entries(payload).filter(([key]) => supportedColumns.includes(key))
      );

      let result;
      if (editId) {
        result = await sb.from('accounts').update(safePayload).eq('id', editId);
      } else {
        result = await sb.from('accounts').insert([safePayload]);
      }

      const alerts = getStoredAccountAlerts();
      const accountId = editId || result?.data?.[0]?.id || payload.id || 'draft';
      alerts[accountId] = {
        min_balance_alert: payload.min_balance_alert ?? null,
        max_balance_alert: payload.max_balance_alert ?? null
      };
      setStoredAccountAlerts(alerts);

      return result || {error: null};
    }
