    function renderMonthlySummary() {
      const card = document.getElementById('monthly-summary');
      const monthLabel = document.getElementById('monthly-summary-month');
      const incomeEl = document.getElementById('monthly-income');
      const expensesEl = document.getElementById('monthly-expenses');
      const remainingEl = document.getElementById('monthly-remaining');
      const selector = document.getElementById('summary-month-select');
      const liveBadge = document.getElementById('summary-live-badge');

      if (!card || !monthLabel || !incomeEl || !expensesEl || !remainingEl) return;

      if (selector && selector.options.length === 0) {
        buildMonthOptions();
      }

      if (selector) {
        selector.value = selectedSummaryMonth;
      }

      if (liveBadge) {
        const now = new Date();
        const currentMonthKey = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`;
        const isCurrentMonth = selectedSummaryMonth === currentMonthKey;
        liveBadge.classList.toggle('hidden', !isCurrentMonth);
      }

      const monthTx = getSelectedMonthTransactions();
      const totalIncome = monthTx
        .filter(tx => parseFloat(tx.amount || 0) > 0)
        .reduce((sum, tx) => sum + parseFloat(tx.amount || 0), 0);

      const totalExpenses = Math.abs(monthTx
        .filter(tx => parseFloat(tx.amount || 0) < 0)
        .reduce((sum, tx) => sum + parseFloat(tx.amount || 0), 0));

      const remaining = totalIncome - totalExpenses;
      const [year, month] = selectedSummaryMonth.split('-').map(Number);
      const monthName = new Date(year, month - 1, 1).toLocaleString('en-US', { month: 'long', year: 'numeric' });

      monthLabel.textContent = monthName;
      incomeEl.textContent = `$${totalIncome.toFixed(2)}`;
      expensesEl.textContent = `$${totalExpenses.toFixed(2)}`;
      remainingEl.textContent = `$${remaining.toFixed(2)}`;
      remainingEl.classList.toggle('text-red-700', remaining < 0);
      remainingEl.classList.toggle('text-blue-700', remaining >= 0);

      card.classList.remove('hidden');
    }

    function renderAllTransactions() {
      const list = document.getElementById('all-transactions-list');
      const searchInput = document.getElementById('transaction-search');
      const query = searchInput ? searchInput.value.trim() : '';

      renderMonthlySummary();

      let baseTransactions = Array.isArray(transactions) ? transactions : [];

      if (selectedSummaryTransactionType === 'income') {
        baseTransactions = baseTransactions.filter(tx => parseFloat(tx.amount || 0) > 0 && !isTransferPartnerTransaction(tx));
      } else if (selectedSummaryTransactionType === 'expense') {
        baseTransactions = baseTransactions.filter(tx => parseFloat(tx.amount || 0) < 0 && !isTransferPartnerTransaction(tx));
      }

      const selectedMonthStart = new Date(`${selectedSummaryMonth}-01T00:00:00`);
      const selectedMonthEnd = new Date(selectedMonthStart.getFullYear(), selectedMonthStart.getMonth() + 1, 0, 23, 59, 59, 999);

      baseTransactions = baseTransactions.filter(tx => {
        const txDate = tx && tx.date ? new Date(tx.date + 'T00:00:00') : null;
        if (!txDate || Number.isNaN(txDate.getTime())) return false;
        return txDate >= selectedMonthStart && txDate <= selectedMonthEnd;
      });

      const filteredTransactions = baseTransactions
        .filter(tx => matchesSearch(tx, query))
        .sort(compareTransactions);

      const groupedTransactions = [];
      const seen = new Set();
      filteredTransactions.forEach(tx => {
        if (seen.has(tx.id)) return;

        if (tx.transfer_id) {
          const partner = filteredTransactions.find(candidate => candidate.transfer_id === tx.transfer_id && candidate.id !== tx.id);
          if (partner) {
            groupedTransactions.push(tx, partner);
            seen.add(tx.id);
            seen.add(partner.id);
            return;
          }
        }

        groupedTransactions.push(tx);
        seen.add(tx.id);
      });

      const finalTransactions = groupedTransactions;

      if (finalTransactions.length === 0) {
        list.innerHTML = `
          <div class="bg-white p-6 rounded-xl border border-gray-200 text-center space-y-2">
            <p class="text-gray-600 font-medium">${query ? 'No matching transactions found.' : selectedSummaryTransactionType === 'income' ? 'No income transactions found for this month.' : selectedSummaryTransactionType === 'expense' ? 'No expense transactions found for this month.' : 'No transactions found.'}</p>
            <p class="text-xs text-gray-400">${query ? 'Try a different search term.' : 'Add a transaction to start tracking your cash flow.'}</p>
          </div>
        `;
        return;
      }

      filteredTransactionsLookup = Object.fromEntries(filteredTransactions.map(tx => [tx.id, tx]));

      const renderTransactionCard = (tx) => {
        const acc = accounts.find(a => a.id === tx.account_id);
        const amount = parseFloat(tx.amount || 0);
        const isPositive = amount >= 0;
        const status = tx.status || 'pending';
        const isPending = status === 'pending';
        const tags = Array.isArray(tx.tags) ? tx.tags : [];
        const isTransfer = !!tx.transfer_id;
        const tagsHtml = tags.length > 0 ? `
          <div class="flex flex-wrap gap-1 mt-1">
            ${tags.map(t => `<span class="text-[10px] bg-blue-50 text-blue-600 px-1.5 py-0.5 rounded font-medium">#${t}</span>`).join('')}
          </div>
        ` : '';

        return `
          <div id="tx-row-${tx.id}" onclick="openTxModal('${tx.id}')" class="${isTransfer ? 'bg-purple-50 border border-purple-200 shadow-sm' : 'bg-white border border-slate-200 shadow-sm'} p-4 rounded-2xl cursor-pointer hover:border-blue-300 transition-all duration-150">
            <div class="flex justify-between items-start gap-3">
              <div class="min-w-0 flex-1 pr-2">
                <div class="flex items-center gap-2">
                  <p class="font-bold text-slate-900 text-base">${tx.payee || 'Transaction'}</p>
                </div>
                <p class="mt-1 text-[11px] text-slate-500 uppercase tracking-[0.14em]">${acc ? acc.name : 'Account'} • ${tx.date || 'No date'}</p>
                ${tx.description && tx.description !== tx.payee ? `<p class="text-xs text-gray-500 mt-0.5">${tx.description}</p>` : ''}
                ${tagsHtml}
                ${renderSplitDetails(tx)}
              </div>
              <div class="flex shrink-0 self-stretch flex-col items-end justify-between gap-2">
                <div class="text-right">
                  <div class="text-lg font-extrabold ${isPositive ? 'text-emerald-600' : 'text-slate-900'}">
                    ${isPositive ? '+' : ''}$${amount.toFixed(2)}
                  </div>
                </div>
                ${renderSplitButton(tx)}
                <div class="flex items-center gap-2">
                  <button 
                    onclick="event.stopPropagation(); toggleTxStatus('${tx.id}', '${status}')" 
                    title="Status: ${status}. Click to toggle"
                    class="px-2 py-0.5 text-xs font-bold rounded-full transition-colors ${isPending ? 'bg-amber-100 text-amber-700 hover:bg-amber-200' : 'bg-emerald-100 text-emerald-700 hover:bg-emerald-200'}">
                    ${isPending ? 'P' : 'R'}
                  </button>
                  <button onclick="event.stopPropagation(); deleteTx('${tx.id}', '${tx.payee || tx.description || 'transaction'}')" class="text-red-400 hover:text-red-600 text-base font-bold px-1">&times;</button>
                </div>
              </div>
            </div>
          </div>
        `;
      };

      const txCardsHtml = [];
      for (let i = 0; i < finalTransactions.length; i++) {
        const tx = finalTransactions[i];
        const nextTx = finalTransactions[i + 1];
        const isPairedTransfer = tx.transfer_id && nextTx && nextTx.transfer_id === tx.transfer_id && nextTx.id !== tx.id;

        if (isPairedTransfer) {
          txCardsHtml.push(`
            <div class="rounded-2xl border border-purple-300 bg-gradient-to-br from-purple-300 via-purple-200 to-purple-100 p-1.5 shadow-sm">
              <div class="space-y-2">
                ${renderTransactionCard(tx)}
                ${renderTransactionCard(nextTx)}
              </div>
            </div>
          `);
          i += 1;
        } else {
          txCardsHtml.push(renderTransactionCard(tx));
        }
      }

      list.innerHTML = txCardsHtml.join('');
    }

    function jumpToTransferPartner(transferId, currentId) {
      const allRows = Array.from(document.querySelectorAll('[id^="tx-row-"]'));
      const partnerRow = allRows.find(row => {
        const rowId = row.id.replace('tx-row-', '');
        const tx = filteredTransactionsLookup?.[rowId];
        return tx && tx.transfer_id === transferId && rowId !== currentId;
      });

      if (partnerRow) {
        partnerRow.scrollIntoView({ behavior: 'smooth', block: 'center' });
        partnerRow.classList.add('ring-2', 'ring-purple-300', 'ring-offset-1');
        setTimeout(() => partnerRow.classList.remove('ring-2', 'ring-purple-300', 'ring-offset-1'), 1800);
      }
    }

    function renderAccounts() {
      renderMonthlySummary();
      const list = document.getElementById('accounts-list');
      const searchInput = document.getElementById('account-search');
      const query = searchInput ? searchInput.value.trim() : '';
      const filteredAccounts = accounts.filter(acc => matchesSearch(acc, query));

      if (filteredAccounts.length === 0) {
        list.innerHTML = `
          <div class="bg-white p-6 rounded-xl border border-gray-200 text-center space-y-2">
            <p class="text-gray-600 font-medium">${query ? 'No matching accounts found.' : 'No accounts found.'}</p>
            <p class="text-xs text-gray-400">${query ? 'Try a different search term.' : 'Tap "+ New Account" at the top to add your checking, savings, or credit card!'}</p>
          </div>
        `;
        return;
      }
      list.innerHTML = filteredAccounts.map(acc => {
        const initialBal = parseFloat(acc.initial_balance || 0);
        // Pull from global transactions array filtered by account ID:

        // Grab transactions from global array, falling back to acc.transactions if empty:
        const globalTxs = (typeof transactions !== 'undefined' && transactions.length > 0)
          ? transactions.filter(tx => (tx.account_id === acc.id || tx.accountId === acc.id))
          : [];

        const txList = globalTxs.length > 0
          ? globalTxs
          : (Array.isArray(acc.transactions) ? acc.transactions : []);

        const totalTxSum = txList.reduce((sum, tx) => sum + parseFloat(tx.amount || 0), 0);
        const currentBal = initialBal + totalTxSum;

        const clearedTxSum = txList.reduce((sum, tx) => {
          return tx.status === 'reconciled' ? sum + parseFloat(tx.amount || 0) : sum;
        }, 0);
        const clearedBal = initialBal + clearedTxSum;

        const {minBalance, maxBalance, minEnabled, maxEnabled, signedBalance, debtBalance, isCreditAccount} = getAccountAlertConfig(acc, currentBal);
        let alertBadge = '';

        if (minEnabled) {
          const minTriggered = isCreditAccount
            ? signedBalance < minBalance
            : signedBalance < minBalance;

          if (minTriggered) {
            alertBadge = `<div class="mt-2 inline-flex items-center rounded-full border border-red-200 bg-red-100 px-2 py-1 text-[10px] font-bold uppercase tracking-wider text-red-700">${isCreditAccount ? 'High Debt' : 'Low Balance'}: $${(isCreditAccount ? Math.abs(minBalance) : minBalance).toFixed(2)}</div>`;
          }
        }

        if (!alertBadge && maxEnabled) {
          const maxTriggered = isCreditAccount
            ? signedBalance > maxBalance
            : signedBalance > maxBalance;

          if (maxTriggered) {
            alertBadge = `<div class="mt-2 inline-flex items-center rounded-full border border-amber-200 bg-amber-100 px-2 py-1 text-[10px] font-bold uppercase tracking-wider text-amber-700">${isCreditAccount ? 'Low Debt' : 'High Balance'}: $${(isCreditAccount ? Math.abs(maxBalance) : maxBalance).toFixed(2)}</div>`;
          }
        }

        const instText = acc.institution ? acc.institution : '';
        const numText = acc.account_number ? `(${acc.account_number})` : '';
        const subDetails = [instText, numText].filter(Boolean).join(' ');

        return `
          <div onclick="showTransactionsView('${acc.id}')" class="bg-white p-4 rounded-2xl border border-slate-200 shadow-sm flex justify-between items-center cursor-pointer transition-all duration-150 hover:-translate-y-0.5 hover:border-blue-300 hover:shadow-md">
            <div class="min-w-0 flex-1 pr-3">
              <p class="font-bold text-slate-900 text-base">${acc.name}</p>
              <p class="mt-1 text-[11px] text-slate-500 uppercase tracking-[0.14em]">${acc.type} ${subDetails ? '• ' + subDetails : ''}</p>
              ${alertBadge}
              <div class="mt-2 text-xs">
                <button onclick="event.stopPropagation(); openAccountModal('${acc.id}')" class="text-blue-600 font-semibold hover:text-blue-700 hover:underline">Edit Account</button>
              </div>
            </div>
            <div class="text-right space-y-1 shrink-0">
              <div>
                <div class="text-lg font-extrabold text-slate-900">$${currentBal.toFixed(2)}</div>
                <div class="text-[10px] text-slate-500 font-semibold uppercase tracking-[0.14em]">Current</div>
              </div>
              <div>
                <div class="text-sm font-bold text-emerald-600">$${clearedBal.toFixed(2)}</div>
                <div class="text-[10px] text-emerald-700 font-semibold uppercase tracking-[0.14em]">Cleared</div>
              </div>
            </div>
          </div>
        `;
      }).join('');
    }

    async function showTransactionsView(accId) {
      window.currentAccountId = accId; // Store active account globally
      activeAccountId = accId;
      selectedTagFilter = null;
      currentViewState = 'transactions-detail';
      const acc = accounts.find(a => a.id === accId);
      if (!acc) return;

      document.getElementById('accounts-view').classList.add('hidden');
      document.getElementById('transactions-view').classList.remove('hidden');

      const instText = acc.institution ? acc.institution : '';
      const numText = acc.account_number ? `(${acc.account_number})` : '';

      document.getElementById('tx-acc-name').textContent = acc.name;
      document.getElementById('tx-acc-sub').textContent = `${acc.type} ${[instText, numText].filter(Boolean).join(' • ')}`;

      await fetchTransactions(accId);
      saveViewState();
    }

    function showAccountsView() {
      window.currentAccountId = null; // Clear active account
      activeAccountId = null;
      selectedTagFilter = null;
      currentViewState = 'accounts-detail';
      document.getElementById('transactions-view').classList.add('hidden');
      document.getElementById('accounts-view').classList.remove('hidden');
      saveViewState();
    }

    async function fetchTransactions(accId) {
      if (!accId) return;
      const list = document.getElementById('transactions-list');
      try {
        const {data, error} = await sb.from('transactions').select('*').eq('account_id', accId)
          .order('status', {ascending: true})
          .order('date', {ascending: false})
          .order('created_at', {ascending: false});
        if (error) throw error;

        currentAccountTransactions = data || [];

        const acc = accounts.find(a => a.id === accId);
        const initialBal = parseFloat(acc ? acc.initial_balance : 0);

        const totalTxSum = currentAccountTransactions.reduce((sum, tx) => sum + parseFloat(tx.amount || 0), 0);
        document.getElementById('tx-acc-balance').textContent = `$${(initialBal + totalTxSum).toFixed(2)}`;

        checkAccountBalanceAlerts();

        const clearedTxSum = currentAccountTransactions.reduce((sum, tx) => {
          return tx.status === 'reconciled' ? sum + parseFloat(tx.amount || 0) : sum;
        }, 0);
        document.getElementById('tx-acc-cleared').textContent = `$${(initialBal + clearedTxSum).toFixed(2)}`;

        renderTagFilters();
        renderTransactions();
      } catch (err) {
        console.error("Error fetching transactions:", err);
        list.innerHTML = `
          <div class="bg-red-50 p-4 rounded-xl border border-red-200 text-center space-y-2">
            <p class="text-red-700 font-semibold text-sm">Error loading transactions</p>
            <p class="text-xs text-red-500">${err.message || 'Check database connection'}</p>
          </div>
        `;
      }
    }

    function renderTagFilters() {
      const filterContainer = document.getElementById('tag-filter-container');
      const chipsContainer = document.getElementById('tag-filter-chips');

      const allTags = new Set();
      currentAccountTransactions.forEach(tx => {
        getTransactionTags(tx).forEach(tag => allTags.add(tag));
      });

      if (allTags.size === 0) {
        filterContainer.classList.add('hidden');
        return;
      }

      filterContainer.classList.remove('hidden');

      let html = `<button onclick="setTagFilter(null)" class="px-2 py-0.5 text-xs font-medium rounded-full ${selectedTagFilter === null ? 'bg-blue-600 text-white' : 'bg-gray-100 text-gray-600 hover:bg-gray-200'
        }">All</button>`;

      allTags.forEach(tag => {
        const isSelected = selectedTagFilter === tag;
        html += `<button onclick="setTagFilter('${tag}')" class="px-2 py-0.5 text-xs font-medium rounded-full ${isSelected ? 'bg-blue-600 text-white' : 'bg-blue-50 text-blue-700 hover:bg-blue-100'
          }">#${tag}</button>`;
      });

      chipsContainer.innerHTML = html;
    }

    function setTagFilter(tag) {
      selectedTagFilter = tag;
      renderTagFilters();
      renderTransactions();
    }

    function escapeHtml(str) {
      if (!str) return '';
      return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
    }

    function renderTransactions() {
      const list = document.getElementById('transactions-list');

      let filteredTx = currentAccountTransactions;
      if (selectedTagFilter) {
        filteredTx = currentAccountTransactions.filter(tx => getTransactionTags(tx).includes(selectedTagFilter));
      }

      const addButton = `
        <button onclick="openTxModal()" class="w-full bg-blue-500 hover:bg-blue-600 text-white font-semibold py-3 rounded-xl transition-colors flex items-center justify-center gap-2 mb-2">
          <span class="text-lg">+</span> Add Transaction
        </button>
      `;

      if (filteredTx.length === 0) {
        list.innerHTML = addButton + `
          <div class="bg-white p-6 rounded-xl border border-gray-200 text-center space-y-2">
            <p class="text-gray-600 font-medium">${selectedTagFilter ? 'No transactions with this tag.' : 'No transactions yet.'}</p>
            <p class="text-xs text-gray-400">${selectedTagFilter ? 'Try picking a different tag filter above.' : 'Tap the button above to log a deposit or spending.'}</p>
          </div>
        `;
        return;
      }

      list.innerHTML = addButton + filteredTx.map(tx => {
        const amt = parseFloat(tx.amount || 0);
        const isPositive = amt >= 0;
        const status = tx.status || 'pending';
        const isPending = status === 'pending';
        const tags = Array.isArray(tx.tags) ? tx.tags : [];
        const isTransfer = !!tx.transfer_id;

        // Main Title: Payer/Payee
        const displayTitle = tx.payee || 'Transaction';

        // Subtitle: ONLY render if description exists AND isn't identical to the payee
        const rawDesc = (tx.description || '').trim();
        const rawPayee = (tx.payee || '').trim();
        const subTitle = (rawDesc && rawDesc.toLowerCase() !== rawPayee.toLowerCase()) ? rawDesc : '';

        const tagsHtml = tags.length > 0 ? `
          <div class="flex flex-wrap gap-1 mt-1">
            ${tags.map(t => `<span class="text-[10px] bg-blue-50 text-blue-600 px-1.5 py-0.5 rounded font-medium">#${t}</span>`).join('')}
          </div>
        ` : '';

        const transferButtonHtml = isTransfer
          ? `
            <button type="button" onclick="event.stopPropagation(); jumpToTransferPartnerAcrossAccounts('${tx.transfer_id}', '${tx.id}', '${activeAccountId || tx.account_id}')" class="inline-flex items-center rounded-full bg-purple-100 text-purple-700 px-1.5 py-0.5 text-[10px] font-bold uppercase tracking-wide hover:bg-purple-200">
              Transfer
            </button>
          `
          : '';

        const receiptUrls = tx.receipt_url ? tx.receipt_url.split(',').map(u => u.trim()).filter(Boolean) : [];
        const receiptBtn = receiptUrls.length > 0 ? `
          <div class="flex gap-1 flex-wrap mt-0.5">
            ${receiptUrls.map((url, idx) => `
              <a href="${url}" target="_blank" onclick="event.stopPropagation()" class="text-[11px] bg-gray-100 hover:bg-gray-200 text-gray-600 px-1.5 py-0.5 rounded flex items-center gap-1 font-medium border border-gray-200">
                📎 Receipt ${receiptUrls.length > 1 ? idx + 1 : ''}
              </a>
            `).join('')}
          </div>
        ` : '';

        return `
          <div id="tx-row-${tx.id}" onclick="openTxModal('${tx.id}')" class="${isTransfer ? 'bg-purple-50 border border-purple-200 shadow-sm' : 'bg-white border border-slate-200 shadow-sm'} p-4 rounded-2xl cursor-pointer hover:border-blue-300 transition-all duration-150">
            <div class="flex justify-between items-start gap-3">
              <div class="min-w-0 flex-1 pr-2">
                <div class="flex items-center gap-2">
                  <p class="font-bold text-slate-900 text-base">${displayTitle}</p>
                </div>
                ${receiptBtn}
                ${subTitle ? `<p class="text-xs text-gray-500 font-medium mt-0.5">${subTitle}</p>` : ''}
                <p class="mt-1 text-[11px] text-slate-500 uppercase tracking-[0.14em]">${tx.date}</p>
                ${tagsHtml}
                ${renderSplitDetails(tx)}
              </div>
              <div class="flex shrink-0 self-stretch flex-col items-end justify-between gap-2">
                <span class="text-lg font-extrabold ${isPositive ? 'text-emerald-600' : 'text-slate-900'}">
                  ${isPositive ? '+' : ''}$${amt.toFixed(2)}
                </span>
                <div class="flex flex-col items-end gap-1">
                  ${transferButtonHtml}
                  ${renderSplitButton(tx)}
                </div>
                <div class="flex items-center gap-2">
                  <button 
                    onclick="event.stopPropagation(); toggleTxStatus('${tx.id}', '${status}')" 
                    title="Status: ${status}. Click to toggle"
                    class="px-2 py-0.5 text-xs font-bold rounded-full transition-colors ${isPending
              ? 'bg-amber-100 text-amber-700 hover:bg-amber-200'
              : 'bg-emerald-100 text-emerald-700 hover:bg-emerald-200'
            }">
                    ${isPending ? 'P' : 'R'}
                  </button>
                  <button onclick="event.stopPropagation(); deleteTx('${tx.id}', '${displayTitle}')" class="text-red-400 hover:text-red-600 text-base font-bold px-1">&times;</button>
                </div>
              </div>
            </div>
          </div>
        `;
      }).join('');
    }

    async function focusTransactionRow(transactionId, attempt = 0) {
      const row = document.getElementById(`tx-row-${transactionId}`);
      if (!row && attempt < 10) {
        await new Promise(resolve => setTimeout(resolve, 50));
        return focusTransactionRow(transactionId, attempt + 1);
      }

      if (!row) return false;

      row.scrollIntoView({ behavior: 'smooth', block: 'center' });
      row.classList.add('ring-2', 'ring-purple-300', 'ring-offset-1');
      setTimeout(() => row.classList.remove('ring-2', 'ring-purple-300', 'ring-offset-1'), 1800);
      return true;
    }

    async function jumpToTransferPartnerAcrossAccounts(transferId, currentId, currentAccountId) {
      const partnerTx = (Array.isArray(transactions) ? transactions : []).find(tx => {
        return tx && tx.transfer_id === transferId && tx.account_id !== currentAccountId && tx.id !== currentId;
      });

      if (partnerTx) {
        if (activeAccountId !== partnerTx.account_id) {
          await showTransactionsView(partnerTx.account_id);
        }
        await focusTransactionRow(partnerTx.id);
        return;
      }

      const partnerTxInCurrentAccount = currentAccountTransactions.find(tx => {
        return tx && tx.transfer_id === transferId && tx.id !== currentId;
      });

      if (partnerTxInCurrentAccount) {
        await focusTransactionRow(partnerTxInCurrentAccount.id);
      }
    }

    async function toggleTxStatus(id, currentStatus) {
      const newStatus = currentStatus === 'pending' ? 'reconciled' : 'pending';
      const {error} = await sb.from('transactions').update({status: newStatus}).eq('id', id);
      if (error) {
        showAppMessage("Error updating status: " + error.message);
        return;
      }
      fetchTransactions(activeAccountId);
      fetchAccounts();
    }

      if (label) {
        label.textContent = type === 'income' ? 'Payer' : 'Payee';
      }
    }

    function handleTypeChange() {
      const type = document.getElementById('txType').value;
      const wrapper = document.getElementById('transferTargetWrapper');
      const paymentSelect = document.getElementById('txPaymentMethod');
      const splitToggle = document.getElementById('txSplitEnabled');

      if (type === 'transfer') {
        wrapper.classList.remove('hidden');
        document.getElementById('tx-account-label').textContent = 'Transfer From Account';
        populateTargetAccountDropdown();
        if (splitToggle) {
          splitToggle.checked = false;
          splitToggle.disabled = true;
        }

        if (paymentSelect) {
          paymentSelect.value = 'bank_transfer';
        }
      } else {
        wrapper.classList.add('hidden');
        document.getElementById('tx-account-label').textContent = 'Account';
        if (splitToggle) splitToggle.disabled = false;
      }

      toggleSplitEditor();

      if (typeof updatePayeeLabel === 'function') {
        updatePayeeLabel();
      }
    }

    function parseTagInput(value) {
      return String(value || '').split(',').map(tag => tag.trim()).filter(Boolean);
    }

    function getTransactionTags(tx) {
      const standardTags = Array.isArray(tx?.tags) ? tx.tags : [];
      const splitTags = Array.isArray(tx?.split_details)
        ? tx.split_details.flatMap(split => Array.isArray(split?.tags)
          ? split.tags
          : parseTagInput(split?.category || ''))
        : [];
      return [...new Set([...standardTags, ...splitTags].map(tag => String(tag).trim()).filter(Boolean))];
    }

    function getSplitRowsFromEditor() {
      return [...document.querySelectorAll('#txSplitRows > div')].map(row => ({
        tags: parseTagInput(row.querySelector('[data-split-tags]')?.value),
        description: row.querySelector('[data-split-description]')?.value.trim() || '',
        amount: parseFloat(row.querySelector('[data-split-amount]')?.value) || 0
      }));
    }

    function renderSplitEditor(rows = [{tags: [], description: '', amount: ''}, {tags: [], description: '', amount: ''}]) {
      const container = document.getElementById('txSplitRows');
      if (!container) return;

      container.innerHTML = rows.map((row, index) => `
        <div class="grid grid-cols-1 gap-2 rounded-lg border border-emerald-100 bg-white p-2 sm:grid-cols-[minmax(0,1fr)_minmax(0,1.2fr)_5rem_auto] sm:items-center">
          <input type="text" data-split-tags placeholder="Tags, comma-separated" value="${escapeHtml((row.tags || (row.category ? [row.category] : [])).join(', '))}"
            class="w-full border border-slate-200 bg-white px-2 py-1.5 rounded-lg text-xs" />
          <input type="text" data-split-description placeholder="Description" value="${escapeHtml(row.description || '')}"
            class="w-full border border-slate-200 bg-white px-2 py-1.5 rounded-lg text-xs" />
          <input type="number" step="0.01" min="0" data-split-amount placeholder="$0.00" value="${row.amount ?? ''}"
            class="w-full border border-slate-200 bg-white px-2 py-1.5 rounded-lg text-xs" />
          <button type="button" onclick="removeSplitRow(${index})" aria-label="Remove split"
            class="text-lg leading-none text-red-400 hover:text-red-600">&times;</button>
        </div>
      `).join('');
      container.querySelectorAll('[data-split-amount]').forEach(input => {
        input.addEventListener('input', () => updateSplitRemainder(input));
      });
      updateSplitRemainder();
    }

    function updateSplitRemainder(changedInput = null) {
      const splitToggle = document.getElementById('txSplitEnabled');
      const rows = [...document.querySelectorAll('#txSplitRows > div')];
      if (!splitToggle?.checked || rows.length < 2) return;

      const amount = Math.abs(parseFloat(document.getElementById('txAmount')?.value) || 0);
      const amountInputs = rows.map(row => row.querySelector('[data-split-amount]'));
      const lastInput = amountInputs[amountInputs.length - 1];
      const previousTotal = amountInputs.slice(0, -1).reduce((sum, input) => sum + (parseFloat(input.value) || 0), 0);
      const remainder = Math.max(0, amount - previousTotal);

      if (changedInput === lastInput) {
        lastInput.dataset.autoRemainder = 'false';
      } else {
        lastInput.value = remainder.toFixed(2);
        lastInput.dataset.autoRemainder = 'true';
      }
    }

    function addSplitRow() {
      const rows = getSplitRowsFromEditor();
      rows.push({tags: [], description: '', amount: ''});
      renderSplitEditor(rows);
    }

    function removeSplitRow(index) {
      const rows = getSplitRowsFromEditor();
      if (rows.length <= 2) return;
      rows.splice(index, 1);
      renderSplitEditor(rows);
    }

    function toggleSplitEditor() {
      const toggle = document.getElementById('txSplitEnabled');
      const editor = document.getElementById('txSplitEditor');
      const standardFields = ['tx-description-field', 'tx-payee-field', 'tx-tags-field'];
      if (!toggle || !editor) return;
      editor.classList.toggle('hidden', !toggle.checked);
      standardFields.forEach(id => document.getElementById(id)?.classList.toggle('hidden', toggle.checked));
      if (toggle.checked && !document.querySelector('#txSplitRows > div')) {
        renderSplitEditor();
      }
    }

    function getSplitDetails(tx) {
      return Array.isArray(tx?.split_details) ? tx.split_details.filter(split => split && Number(split.amount) > 0) : [];
    }

    function toggleSplitDetails(transactionId) {
      if (splitDetailsExpanded.has(transactionId)) {
        splitDetailsExpanded.delete(transactionId);
      } else {
        splitDetailsExpanded.add(transactionId);
      }
      renderAllTransactions();
      if (activeAccountId) renderTransactions();
    }

    function renderSplitButton(tx) {
      const splits = getSplitDetails(tx);
      if (splits.length === 0) return '';
      return `
        <button type="button" onclick="event.stopPropagation(); toggleSplitDetails('${tx.id}')"
          class="inline-flex items-center rounded-full bg-emerald-100 px-1.5 py-0.5 text-[10px] font-bold uppercase tracking-wide text-emerald-700 hover:bg-emerald-200">
          ${splitDetailsExpanded.has(tx.id) ? 'Hide Split' : `Split (${splits.length})`}
        </button>
      `;
    }

    function renderSplitDetails(tx) {
      const splits = getSplitDetails(tx);
      if (splits.length === 0 || !splitDetailsExpanded.has(tx.id)) return '';
      return `
          <div class="mt-2 space-y-1 rounded-xl border border-blue-100 bg-blue-50 p-2">
            ${splits.map(split => `
              <div class="flex items-center justify-between gap-2 text-xs">
                <span class="min-w-0 text-slate-700">
                  <span class="block font-semibold">${escapeHtml((split.tags || (split.category ? [split.category] : [])).join(', ') || 'Untagged')}</span>
                  ${split.description ? `<span class="block text-[11px] text-slate-500">${escapeHtml(split.description)}</span>` : ''}
                </span>
                <span class="font-bold text-slate-900">$${Number(split.amount).toFixed(2)}</span>
              </div>
            `).join('')}
          </div>
      `;
    }
